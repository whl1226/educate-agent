## Task 1: agent 页自动发送后输入框保持空白

**根因**：`agent-main.ts:275-288` 把 pendingTask 写入 `input.value` 后再 `sendTask(input.value)`，流式执行期间输入框一直显示任务文本（用户看到"重复"）。

**Files:**
- Modify: `src/frontend/entries/agent-main.ts:267-288`

- [ ] **Step 1: 改写自动发送段（不写入输入框，直接发送；完成后输入框保持空白并聚焦）**

```ts
  // ===== 教师端「发送给智能体」任务：sessionStorage 传递 → 自动发送 =====
  let pendingTask: string | null = null;
  try {
    pendingTask = sessionStorage.getItem(TASK_CHANNEL);
    if (pendingTask) sessionStorage.removeItem(TASK_CHANNEL);
  } catch {
    /* 静默 */
  }
  if (pendingTask && pendingTask.trim()) {
    const task = pendingTask.trim();
    window.setTimeout(() => {
      const welcome = document.getElementById('welcomeBox');
      if (welcome) welcome.style.display = 'none';
      controller?.switchView('chat');
      // 输入框不填充任务文本：任务已由 sessionStorage 传递，直接发送
      void controller!.sendTask(task).finally(() => {
        input.value = '';
        autoGrow(input);
        input.focus();
      });
    }, 120);
  }
```

```ts
const GREEK: Record<string, string> = {
  alpha: 'α', beta: 'β', gamma: 'γ', delta: 'δ', epsilon: 'ε', theta: 'θ',
  lambda: 'λ', mu: 'μ', pi: 'π', rho: 'ρ', sigma: 'σ', phi: 'φ', omega: 'ω',
  Omega: 'Ω', Delta: 'Δ', Sigma: 'Σ',
};

const OPS: Record<string, string> = {
  times: '×', div: '÷', pm: '±', mp: '∓', cdot: '·', approx: '≈',
  neq: '≠', leq: '≤', geq: '≥', ll: '≪', gg: '≫', to: '→', in: '∈',
  notin: '∉', subset: '⊂', supset: '⊃', cap: '∩', cup: '∪', forall: '∀',
  exists: '∃', partial: '∂', nabla: '∇', infty: '∞', angle: '∠',
  parallel: '∥', perp: '⊥', degree: '°', percent: '%', prime: '′',
};

const SUPER: Record<string, string> = {
  '0': '⁰', '1': '¹', '2': '²', '3': '³', '4': '⁴', '5': '⁵',
  '6': '⁶', '7': '⁷', '8': '⁸', '9': '⁹', '+': '⁺', '-': '⁻',
  'n': 'ⁿ', 'x': 'ˣ', 'a': 'ᵃ', 'm': 'ᵐ',
};

const SUB: Record<string, string> = {
  '0': '₀', '1': '₁', '2': '₂', '3': '₃', '4': '₄', '5': '₅',
  '6': '₆', '7': '₇', '8': '₈', '9': '₉', '+': '₊', '-': '₋',
  'a': 'ₐ', 'x': 'ₓ', 'i': 'ᵢ', 'n': 'ₙ',
};

function asciiToUni(ch: string, map: Record<string, string>): string {
  return map[ch] ?? ch;
}

/** 行内公式 $...$ 与块级 $$...$$ → Unicode 文本 */
export function latexToUnicode(raw: string): string {
  let s = String(raw ?? '');
  // 块级 $$...$$ → 保留换行包裹
  s = s.replace(/\$\$([\s\S]*?)\$\$/g, (_m, body: string) => `\n${convert(body)}\n`);
  // 行内 $...$
  s = s.replace(/\$([^$\n]+)\$/g, (_m, body: string) => convert(body));
  // 兜底：清理孤立 $ 符号（防止残留破坏排版）
  s = s.replace(/\$/g, '');
  return s;
}

/** 单个公式体转换（LaTeX 子集） */
export function convert(body: string): string {
  let s = body.trim();
  // 分数：\frac{a}{b} → (a)/(b)；嵌套用递归式（两级足够）
  s = s.replace(/\\frac\{([^{}]*)\}\{([^{}]*)\}/g, (_m, a: string, b: string) => `(${convert(a)})/(${convert(b)})`);
  // 根式：\sqrt{a} → √(a)
  s = s.replace(/\\sqrt(?:\[([^\]]*)\])?\{([^{}]*)\}/g, (_m, idx: string | undefined, a: string) => `√[${idx ?? ''}](${convert(a)})`);
  // 上下标：x^{2} / x^2 / x_{1} / x_1
  s = s.replace(/\^\{([^{}]*)\}/g, (_m, a: string) => `^${convert(a)}`);
  s = s.replace(/_\{([^{}]*)\}/g, (_m, a: string) => `_${convert(a)}`);
  s = s.replace(/\^([0-9a-zA-Z+\-])/g, (_m, a: string) => SUPER[a] ?? `^${a}`);
  s = s.replace(/_([0-9a-zA-Z+\-])/g, (_m, a: string) => SUB[a] ?? `_${a}`);
  // 上标块（x^{2} 已转成 ^2，再统一转 Unicode；含字母保留 ^ 记号）
  s = s.replace(/\^([0-9]+)/g, (_m, d: string) => d.split('').map((c) => SUPER[c] ?? c).join(''));
  // 希腊字母
  for (const [k, v] of Object.entries(GREEK)) {
    s = s.replace(new RegExp(`\\\\${k}(?![a-zA-Z])`, 'g'), v);
  }
  // 运算符
  for (const [k, v] of Object.entries(OPS)) {
    s = s.replace(new RegExp(`\\\\${k}(?![a-zA-Z])`, 'g'), v);
  }
  // 空格命令 \quad \qquad \, \;
  s = s.replace(/\\quad|\\qquad|\\,|\\;/g, ' ');
  // 花括号文本 {abc} → abc
  s = s.replace(/\{([^{}]*)\}/g, '$1');
  // 残余反斜杠命令（未知命令降级为去反斜杠）
  s = s.replace(/\\([a-zA-Z]+)/g, '$1');
  // 去掉 ^ _ 残留（未知上标下标）
  s = s.replace(/[\^_]/g, '');
  // 圆括号美化：连续的 (x)/(y) → x/y（仅当分子分母为简单量）
  s = s.replace(/\(([0-9a-zA-Zα-ωπ√²³]+)\)\s*\/\s*\(([0-9a-zA-Zα-ωπ√²³]+)\)/g, '$1/$2');
  return s.trim();
}

/** 供 md-validator/渲染器统一入口：文本先公式转换再剥离 md 符号 */
export function stripMdAndEmojiWithLatex(raw: string): string {
  return latexToUnicode(raw);
}
```

```ts
      case 'table': {
        const cellPad = 4;
        const colCount = Math.max(block.headers.length, 1);
        const colWidth = (maxWidth - 2 * margin) / colCount;
        const cellFontSize = 9;
        const lineHeight = cellFontSize + 3;
        const wrapCell = (text: string): string[] => {
          const lines: string[] = [];
          let line = '';
          for (const ch of stripMdAndEmoji(text)) {
            if (font.widthOfTextAtSize(line + ch, cellFontSize) > colWidth - 2 * cellPad - 2) {
              lines.push(line);
              line = ch;
            } else {
              line += ch;
            }
          }
          if (line) lines.push(line);
          return lines.length ? lines : [''];
        };
        const rowOf = (cells: string[], header: boolean) => {
          const wrapped = cells.slice(0, colCount).map(wrapCell);
          const maxLines = Math.max(...wrapped.map((l) => l.length), 1);
          const rowHeight = maxLines * lineHeight + 6;
          if (cur < bottom + rowHeight + 8) {
            pdf.addPage([595.28, 841.89]);
            cur = height - 60;
          }
          for (let i = 0; i < colCount; i++) {
            page.drawRectangle({
              x: x + margin + i * colWidth, y: cur - rowHeight,
              width: colWidth - 2, height: rowHeight - 2,
              color: header ? COLOR_TABLE_HEADER_BG : COLOR_TABLE_CELL_BG,
            });
          }
          for (let i = 0; i < colCount; i++) {
            const cellX = x + margin + i * colWidth;
            wrapped[i].forEach((ln, li) => {
              page.drawText(ln, {
                x: cellX + cellPad, y: cur - cellFontSize - 2 - li * lineHeight,
                size: cellFontSize, font,
                color: header ? COLOR_WHITE : COLOR_TABLE_CELL_TEXT,
              });
            });
          }
          cur -= rowHeight + 2;
        };
        rowOf(block.headers, true);
        for (const row of block.rows) rowOf(row, false);
        return cur;
      }
```

```ts
/** 智能体生成试卷后回到本页：预览区展示最新试卷（若无则保留引导文案） */
async function refreshPaperPreview() {
  try {
    const papers = await get<any[]>('/papers');
    if (!papers || !papers.length) {
      ['paperDlDocx', 'paperDlPdf'].forEach((id) => {
        const b = document.getElementById(id) as HTMLButtonElement | null;
        if (b) b.disabled = true;
      });
      return;
    }
    ['paperDlDocx', 'paperDlPdf'].forEach((id) => {
      const b = document.getElementById(id) as HTMLButtonElement | null;
      if (b) b.disabled = false;
    });
    const previewCard = findCardByTitle('#page-paper', '试卷预览');
    if (!previewCard) return;
    const listBox = previewCard.querySelector('.card-head + div');
    const paper = papers[0];
    if (listBox) {
      const rows = (paper.questions ?? []).length
        ? (paper.questions as Array<{ stem: string; layer: string; answer: string; analysis?: string }>)

      listBox.innerHTML = rows;
```

```ts
export function renderResearcherDims(body: HTMLElement, res: { score?: number; dims?: Array<{ name: string; score: number; comment: string }>; mustFix?: string[]; suggest?: string[]; standard?: string }): void {
  const dims = res.dims ?? [];
  body.innerHTML =
    `<div class="grid g2" style="gap:10px;margin-bottom:14px">` +
    dims.map((d) => `<div style="background:var(--primary-soft);border-radius:10px;padding:10px 12px">
        <div style="display:flex;justify-content:space-between;align-items:center">
          <span style="font-size:12.5px;font-weight:700">${esc(d.name)}</span>
          <span class="mono" style="font-size:15px;color:var(--primary)">${escNum(d.score)}</span>
        </div>
        <div style="font-size:11.5px;color:var(--muted);margin-top:4px;line-height:1.6">${esc(d.comment)}</div>
      </div>`).join('') +
    `</div>` +
    (res.mustFix?.length ? `<div style="background:#fde8ec;border:1px solid #fecdd3;border-radius:10px;padding:10px 14px;margin-bottom:10px"><div style="font-size:11.5px;font-weight:700;color:#be123c;margin-bottom:5px">必须改进</div>${res.mustFix.map((s) => `<div style="font-size:12.5px;color:#4b5162;line-height:1.8">· ${esc(s)}</div>`).join('')}</div>` : '') +
    (res.suggest?.length ? `<div style="background:#fffbf0;border:1px solid #fde68a;border-radius:10px;padding:10px 14px;margin-bottom:10px"><div style="font-size:11.5px;font-weight:700;color:#b45309;margin-bottom:5px">建议优化</div>${res.suggest.map((s) => `<div style="font-size:12.5px;color:#4b5162;line-height:1.8">· ${esc(s)}</div>`).join('')}</div>` : '') +

}
```

```ts
export function demoSkillReport(self: Record<string, number>) {
  const radar = { 教学设计: self['教学设计'] ?? 82, 课堂管理: self['课堂管理'] ?? 74, 粉笔字: self['粉笔字'] ?? 68, 普通话: self['普通话'] ?? 86, 信息化: self['信息化'] ?? 63, 学情分析: self['学情分析'] ?? 78, 作业设计: self['作业设计'] ?? 70, 家校沟通: self['家校沟通'] ?? 75 };
  const weakest = Object.entries(radar).sort((a, b) => a[1] - b[1]).slice(0, 3);
  const plan = weakest.map(([k, v], i) => ({
    item: k,
    score: v,
    days: `第 ${i * 2 + 1}-${i * 2 + 2} 天`,
    train: `针对「${k}」：第 1 天完成 2 次微格练习并录音自评；第 2 天观摩同组优秀案例 1 次并写 300 字复盘；每项配 15 分钟练习 + 自评表（1-5 分）。`,
    resource: k === '信息化' ? '推荐：国家中小学智慧教育平台·课件制作微课' : k === '粉笔字' ? '推荐：每日 15 分钟楷书结构临摹 + 拍照对比' : '推荐：教研规则库对应训练模块',
  }));
  return { radar: JSON.stringify(radar), plan: JSON.stringify(plan) };
}
```
