- [ ] **Step 4: 实现 extractText**

`files.service.ts` 追加：

```ts
import * as mammoth from 'mammoth';
import * as pdfParse from 'pdf-parse';

/** 按魔数识别 docx/pdf 并提取纯文本（供 AI 教研员解析上传材料） */
async extractText(buf: Buffer, filename: string): Promise<string> {
  const isDocx = buf.length >= 4 && buf[0] === 0x50 && buf[1] === 0x4b && buf[2] === 0x03 && buf[3] === 0x04;
  const isPdf = buf.subarray(0, 5).toString('latin1') === '%PDF-';
  if (isDocx) {
    const r = await mammoth.extractRawText({ buffer: buf });
    return (r.value || '').slice(0, 60000);
  }
  if (isPdf) {
    const r = await pdfParse(buf);
    return (r.text || '').slice(0, 60000);
  }
  throw new BizException(ErrorCodes.VALIDATE_ERROR, '仅支持 Word(.docx) 或 PDF 文件');
}
```

（BizException/ErrorCodes 以 files 模块现有引用路径为准，若无则用模块内现有错误抛法。）

```ts
private readonly LICENSE_WHITELIST = ['自建', '共享', '公开领域', 'CC BY'];

async listResources(user: { id: number; role: string }, query: { license?: string; q?: string; type?: string }) {
  const qb = this.resourcesRepo.createQueryBuilder('r')
    .where('r.teacherId = :uid OR r.license IN (:...open)', {
      uid: user.id,
      open: ['公开领域', 'CC BY', '共享'],
    });
  if (query.license && this.LICENSE_WHITELIST.includes(query.license)) qb.andWhere('r.license = :lic', { lic: query.license });
  if (query.type) qb.andWhere('r.type = :type', { type: query.type });
  if (query.q) qb.andWhere('(r.title LIKE :q OR r.description LIKE :q)', { q: `%${query.q}%` });
  return qb.orderBy('r.createdAt', 'DESC').getMany();
}

async createResource(user: { id: number; role: string }, dto: { title: string; type: string; license: string; description?: string; url?: string }) {
  if (!this.LICENSE_WHITELIST.includes(dto.license)) throw new BizException(ErrorCodes.VALIDATE_ERROR, '授权范围非法');
  const entity = this.resourcesRepo.create({
    teacherId: user.id, title: dto.title, type: dto.type, license: dto.license,
    description: dto.description ?? '', url: dto.url ?? '',
  });
  return this.resourcesRepo.save(entity);
}
```

```ts
const RESOURCE_TYPES: Record<string, string> = {
  教案: '教案', 课件: '课件', 试卷: '习题', '教学建议': '其他', '发言稿': '其他', '开学材料': '其他', '话术': '其他',
};

/** 生成类交付物自动归档到教学资源库（自建授权） */
export async function autoStoreResource(
  ctx: { resourcesRepo: any; userId: number },
  title: string, type: string, text: string,
) {
  const mapped = RESOURCE_TYPES[type];
  if (!mapped || !title || !text) return null;
  return ctx.resourcesRepo.save(
    ctx.resourcesRepo.create({
      teacherId: ctx.userId, title: title.slice(0, 80), type: mapped, license: '自建',
      description: (text || '').slice(0, 200), url: '',
    }),
  );
}
```

```ts
  if (!runId) return { messages: [] as Array<{ role: string; content: string }>, title: null as string | null };
```

```ts
async groupDetail(user: { id: number; role: string }, id: number) {
  const g = await this.collabGroupsRepo.findOne({ where: { id } });
  if (!g) throw new BizException(ErrorCodes.NOT_FOUND);
  if (user.role !== 'admin' && !this.isCollabMember(g, user.id)) throw new BizException(ErrorCodes.SCOPE_FORBIDDEN);
  const feeds = await this.collabFeedsRepo.find({ where: { groupId: id }, order: { id: 'ASC' } });
  return { ...g, feeds: feeds.map((f) => ({ id: f.id, kind: f.kind, payload: safeJson(f.content), at: f.createdAt })) };
}

async addCollabComment(user: { id: number; role: string }, groupId: number, dto: { text: string; target?: string; authorName?: string }) {
  const g = await this.collabGroupsRepo.findOne({ where: { id: groupId } });
  if (!g) throw new BizException(ErrorCodes.NOT_FOUND);
  if (user.role !== 'admin' && !this.isCollabMember(g, user.id)) throw new BizException(ErrorCodes.SCOPE_FORBIDDEN);
  if (!dto.text || !dto.text.trim()) throw new BizException(ErrorCodes.VALIDATE_ERROR, '批注内容必填');
  const feed = this.collabFeedsRepo.create({
    groupId, kind: 'comment', teacherId: user.id,
    content: JSON.stringify({ type: 'comment', author: dto.authorName || ('教师' + user.id), target: dto.target ?? '', text: dto.text.slice(0, 2000), status: 'open', at: Date.now() }),
  });
  return this.collabFeedsRepo.save(feed);
}

async adoptCollabComment(user: { id: number; role: string }, feedId: number) {
  const feed = await this.collabFeedsRepo.findOne({ where: { id: feedId } });
  if (!feed) throw new BizException(ErrorCodes.NOT_FOUND);
  if (user.role !== 'admin' && feed.teacherId !== user.id) throw new BizException(ErrorCodes.SCOPE_FORBIDDEN);
  const payload = safeJson(feed.content);
  payload.status = 'adopted';
  feed.content = JSON.stringify(payload);
  return this.collabFeedsRepo.save(feed);
}
```

- 新增依赖只装在 server（mammoth/pdf-parse），前端不引第三方运行时
- 涉及实体新增字段时先跑 `npm run seed` 或确认 TypeORM synchronize 开启（主 app 配置为准）；若 synchronize 关闭，需补迁移语句（按 server/src/db 现有迁移机制）
- Task 16 模拟批注仅前端会话内可见；真实评论走 API 持久化——验收时两者都能看到
- P0 完成后可先交付验证，再进入 P1
