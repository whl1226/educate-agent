import { inflateRawSync } from 'node:zlib';

function zipEntries(buf: Buffer): { name: string; data: Buffer }[] {
  let eocd = -1;
  for (let i = buf.length - 22; i >= 0; i--) {
    if (buf.readUInt32LE(i) === 0x06054b50) { eocd = i; break; }
  }
  const count = buf.readUInt16LE(eocd + 10);
  const cdOffset = buf.readUInt32LE(eocd + 16);
  const entries: { name: string; data: Buffer }[] = [];
  let pos = cdOffset;
  for (let n = 0; n < count; n++) {
    const method = buf.readUInt16LE(pos + 10);
    const compSize = buf.readUInt32LE(pos + 20);
    const nameLen = buf.readUInt16LE(pos + 28);
    const extraLen = buf.readUInt16LE(pos + 30);
    const commentLen = buf.readUInt16LE(pos + 32);
    const localOff = buf.readUInt32LE(pos + 42);
    const name = buf.toString('utf8', pos + 46, pos + 46 + nameLen);
    const lhNameLen = buf.readUInt16LE(localOff + 26);
    const lhExtraLen = buf.readUInt16LE(localOff + 28);
    const dataStart = localOff + 30 + lhNameLen + lhExtraLen;
    const raw = buf.subarray(dataStart, dataStart + compSize);
    entries.push({ name, data: method === 8 ? inflateRawSync(raw) : Buffer.from(raw) });
    pos += 46 + nameLen + extraLen + commentLen;
  }
  return entries;
}

it('同页多个块不重叠（y 坐标唯一）', async () => {

```ts
import PptxGenJS from 'pptxgenjs';
import type { DocumentRenderer, OfficeBlock, ParsedDoc } from '../office.types';
import { themeOf } from '../themes';

const W = 13.33;
const H = 7.5;
const MARGIN = 0.85;
const FOOTER_Y = 7.05;          // 内容不可越过 footer
const START_Y = 1.2;

export class PptxRenderer implements DocumentRenderer {
  readonly format = 'pptx' as const;

  /** 估算块高度（英寸） */
  private blockHeight(block: OfficeBlock): number {
    switch (block.type) {
      case 'heading': return block.level === 1 ? 0.9 : 0.6;
      case 'paragraph': return 0.7;
      case 'list': return Math.min(block.items.length * 0.5, 3.5);
      case 'code': return Math.min(1 + block.code.split('\n').length * 0.25, 3.5);
      case 'table': return Math.min(0.5 + block.rows.length * 0.35, 4.5);
    }
  }

  /** 新页：页眉标题 + 顶部 accent 条 + 页脚页码 */
  private newSlide(pptx: PptxGenJS, title: string, themeKey: string, pageNo: number): PptxGenJS.Slide {
    const t = themeOf(themeKey);
    const slide = pptx.addSlide();
    slide.addShape(pptx.shapes.RECTANGLE, { x: 0, y: 0, w: W, h: 0.06, fill: { color: t.palette.accent } });
    slide.addText(title, { x: MARGIN, y: 0.3, w: 8, h: 0.5, fontSize: t.scale.title, bold: true, color: t.palette.accent });
    slide.addText(String(pageNo), { x: W - MARGIN - 1, y: H - 0.5, w: 1, h: 0.35, fontSize: t.scale.small, color: t.palette.muted, align: 'right' });
    return slide;
  }

  private addBlock(slide: PptxGenJS.Slide, block: OfficeBlock, themeKey: string, y: number): number {
    const t = themeOf(themeKey);
    const w = W - 2 * MARGIN;
    switch (block.type) {
      case 'heading': {
        const size = block.level === 1 ? t.scale.title : Math.max(18, t.scale.body + 2);
        slide.addText(block.text, { x: MARGIN, y, w, h: 0.6, fontSize: size, bold: true, color: block.level === 1 ? t.palette.accent : t.palette.accent2 });
        return y + 0.6;
      }
      case 'paragraph':
        slide.addText(block.text, { x: MARGIN, y, w, h: 0.7, fontSize: t.scale.body, color: t.palette.text, valign: 'top', breakLine: true });
        return y + 0.7;
      case 'list':
        slide.addText(
          block.items.map((item) => ({ text: item, options: { bullet: { characterCode: '25CF' } } })),
          { x: MARGIN + 0.15, y, w: w - 0.15, h: Math.min(block.items.length * 0.5, 3.5), fontSize: t.scale.body, color: t.palette.text, valign: 'top' },
        );
        return y + Math.min(block.items.length * 0.5, 3.5);
      case 'code':
        slide.addText(block.code, { x: MARGIN, y, w, h: Math.min(1 + block.code.split('\n').length * 0.25, 3.5), fontSize: 12, fontFace: 'Consolas', color: '1a1a1a', fill: { color: 'F4F4F4' } });
        return y + Math.min(1 + block.code.split('\n').length * 0.25, 3.5);
      case 'table': {
        const maxRows = Math.floor((FOOTER_Y - y) / 0.35) - 1;
        const visible = block.rows.slice(0, Math.max(1, maxRows));
        const rows: PptxGenJS.TableRow[] = [
          block.headers.map((h) => ({ text: h, options: { fill: { color: t.palette.accent }, color: 'FFFFFF', bold: true } })),
          ...visible.map((row) => row.map((cell) => ({ text: cell }))),
        ];
        slide.addTable(rows, { x: MARGIN, y, w, border: { pt: 0.5, color: t.palette.accent2 } });
        if (block.rows.length > visible.length) {
          slide.addText(`… 其余 ${block.rows.length - visible.length} 行见后续页`, { x: MARGIN, y: FOOTER_Y - 0.5, w, h: 0.35, fontSize: t.scale.small, color: t.palette.muted });
        }
        return y + 0.5 + visible.length * 0.35;
      }
    }
  }

  async render(doc: ParsedDoc): Promise<Buffer> {
    const pptx = new PptxGenJS();
    const t = themeOf(doc.theme);
    pptx.layout = 'LAYOUT_16x9';
    pptx.author = doc.author ?? '乡芽';

    // 封面页
    const cover = pptx.addSlide();
    cover.addShape(pptx.shapes.RECTANGLE, { x: 0, y: 0, w: W, h: H, fill: { color: t.palette.accent } });
    cover.addText(doc.title, { x: 0.8, y: 2.0, w: 11.7, h: 1.4, fontSize: t.scale.hero, bold: true, color: 'FFFFFF', align: 'center' });
    if (doc.author) cover.addText(`编制人：${doc.author}`, { x: 0.8, y: 3.6, w: 11.7, h: 0.6, fontSize: 18, color: 'FFFFFF', align: 'center', transparency: 20 });

    // 内容页：垂直游标 + 超页开新页
    let slide: PptxGenJS.Slide = pptx.addSlide();
    let y = START_Y;
    let page = 1;
    const pushSlide = () => {
      slide = this.newSlide(pptx, doc.title, doc.theme, ++page);
      y = START_Y;
    };
    for (const b of doc.blocks) {
      if (b.type === 'heading' && b.level === 1) {
        pushSlide();
        y = this.addBlock(slide, b, doc.theme, y);
        continue;
      }
      const h = this.blockHeight(b);
      if (y + h > FOOTER_Y) pushSlide();
      y = this.addBlock(slide, b, doc.theme, y);
    }
    return Buffer.from((await pptx.write({ outputType: 'nodebuffer' })) as Uint8Array);
  }
}
```

- [ ] **Step 4: 创建 `question-bank-validator.ts`**（复用 office.types 的 ValidationIssue）

```ts
import type { ValidationIssue } from '../office.types';
import type { QuestionBank, QuestionType } from './question-bank.types';

const QUESTION_TYPES: QuestionType[] = ['choice', 'fill', 'judge', 'compute', 'answer'];
const MAX_QUESTIONS = 50;
const MAX_STEM = 2000;
const MAX_ANSWER = 500;
const MAX_OPTIONS = 6;

export function validateQuestionBank(bank: unknown): { valid: boolean; issues: ValidationIssue[] } {
  const issues: ValidationIssue[] = [];
  const b = bank as QuestionBank;
  if (!b || typeof b !== 'object') {
    issues.push({ code: 'QB_NOT_OBJECT', line: 1, message: '题集必须是对象', fix: '返回 {"paperTitle":"...","questions":[...]}' });
    return { valid: false, issues };
  }
  if (typeof b.paperTitle !== 'string' || !b.paperTitle.trim() || b.paperTitle.length > 80) {
    issues.push({ code: 'QB_TITLE', line: 1, message: 'paperTitle 必填且 ≤80 字', fix: '添加试卷名称' });
  }
  if (!Array.isArray(b.questions) || b.questions.length === 0) {
    issues.push({ code: 'QB_EMPTY', line: 1, message: 'questions 为空', fix: '至少提取 1 道题' });
    return { valid: issues.length === 0, issues };
  }
  if (b.questions.length > MAX_QUESTIONS) {
    issues.push({ code: 'QB_TOO_MANY', line: 1, message: `题目数超过 ${MAX_QUESTIONS}`, fix: '按题号分批生成' });
  }
  b.questions.forEach((q, idx) => {
    const line = idx + 1;
    const at = (code: string, message: string, fix: string) => issues.push({ code, line, message, fix });
    if (!q || typeof q !== 'object') return at('QB_Q_NOT_OBJECT', '题目必须是对象', '检查题目结构');
    if (q.no !== idx + 1) return at('QB_NO_SEQUENCE', `题号应为 ${idx + 1}，实际 ${q.no}`, '题号从 1 开始连续编号');
    if (!QUESTION_TYPES.includes(q.type)) return at('QB_BAD_TYPE', `type 必须是 ${QUESTION_TYPES.join('/')}`, '修正题目类型');
    if (typeof q.stem !== 'string' || !q.stem.trim()) return at('QB_EMPTY_STEM', '题干为空', '补全题干');
    if (q.stem.length > MAX_STEM) return at('QB_STEM_TOO_LONG', `题干超过 ${MAX_STEM} 字`, '精简题干');
    if (q.type === 'choice') {
      if (!Array.isArray(q.options) || q.options.length < 2 || q.options.length > MAX_OPTIONS) {
        return at('QB_OPTIONS_REQUIRED', `选择题必须有 2-${MAX_OPTIONS} 个选项`, '补齐选项');
      }
      const letters = 'ABCDEF';
      const optsOk = q.options.every((o, i) => typeof o === 'string' && o.trim() && letters[i] !== undefined);
      if (!optsOk) return at('QB_OPTIONS_BAD', '选项必须是字符串数组', '检查选项格式');
      if (typeof q.answer !== 'string' || !/^[A-F]$/.test(q.answer.trim())) {
        return at('QB_ANSWER_LETTER', '选择题答案必须是 A-F 单个字母', '修正答案');
      }
      if (q.answer.trim().charCodeAt(0) - 65 >= q.options.length) {
        return at('QB_ANSWER_OUT_OF_RANGE', '答案字母超出选项数', '修正答案');
      }
    } else {
      if (typeof q.answer !== 'string' || !q.answer.trim() || q.answer.length > MAX_ANSWER) {
        return at('QB_ANSWER_REQUIRED', '答案必填且 ≤500 字', '补全答案');
      }
    }
    if (q.points !== undefined && (typeof q.points !== 'number' || q.points < 0 || q.points > 20)) {
      at('QB_POINTS', '分值必须在 0-20', '修正分值');
    }
  });
  return { valid: issues.length === 0, issues };
}

  async render(bank: QuestionBank, themeKey: string): Promise<Buffer> {
    const pptx = new PptxGenJS();
    const t = themeOf(themeKey);
    pptx.layout = 'LAYOUT_16x9';

import { Injectable, Logger } from '@nestjs/common';
import pdfParse from 'pdf-parse';
import { createWorker } from 'tesseract.js';

export type ExamTextSource = 'plain' | 'pdf' | 'ocr';

export interface ExamTextResult {
  source: ExamTextSource;
  text: string;
  confidence?: number;
}

@Injectable()
export class ExamTextService {
  private readonly logger = new Logger(ExamTextService.name);

  /** 按输入类型分派解析。text 入参为纯文本字符串；buffer 入参配合 type 解析文件 */
  async extract(input: { text?: string } | { buffer: Buffer; type: 'pdf' | 'image' }, declared?: ExamTextSource): Promise<ExamTextResult> {
    if ('text' in input && typeof input.text === 'string' && input.text.trim()) {
      return { source: 'plain', text: input.text.trim() };
    }
    if (!('buffer' in input)) throw new Error('输入为空：请粘贴卷子文字或上传文件');
    const type = declared ?? 'pdf';
    if (type === 'pdf') return this.fromPdf(input.buffer);
    if (type === 'image') return this.fromImage(input.buffer);
    throw new Error('不支持的文件类型（仅支持 PDF / 图片 / 文本）');
  }

  /** PDF → 文本层提取（扫描版 PDF 无文本层会得到空串，由调用方降级提示） */
  async fromPdf(buffer: Buffer): Promise<ExamTextResult> {
    try {
      const data = await pdfParse(buffer);
      const text = (data.text ?? '').trim();
      if (!text) throw new Error('PDF 无文本层（可能是扫描件），请拍照上传或粘贴文字');
      return { source: 'pdf', text };
    } catch (e) {
      this.logger.warn(`pdf 提取失败: ${(e as Error).message}`);
      throw e;
    }
  }

  /** 图片 → Tesseract.js OCR（chi_sim+eng）。识别失败/模型缺失时抛错，由 Agent 引导粘贴文本 */
  async fromImage(buffer: Buffer): Promise<ExamTextResult> {
    let worker: Awaited<ReturnType<typeof createWorker>> | null = null;
    try {
      worker = await createWorker(['chi_sim', 'eng'], 1, { logger: undefined });
      const { data } = await worker.recognize(buffer);
      const text = (data.text ?? '').trim();
      if (!text) throw new Error('OCR 未识别到文字，请确认图片清晰（印刷体）或直接粘贴文字');
      if ((data.confidence ?? 0) < 50) {
        this.logger.warn(`OCR 置信度低: ${data.confidence}`);
      }
      return { source: 'ocr', text, confidence: data.confidence };
    } catch (e) {
      this.logger.warn(`OCR 失败: ${(e as Error).message}`);
      throw new Error(`图片识别失败（${(e as Error).message}），建议粘贴卷子文字或上传清晰的印刷体图片`);
    } finally {
      try { await worker?.terminate(); } catch { /* ignore */ }
    }
  }
}

  if (!/^[\w.-]+$/.test(storedName)) throw new BizException(ErrorCodes.NOT_FOUND, '文件不存在');
  const abs = join(this.uploadDir(), storedName);
  if (!existsSync(abs)) throw new BizException(ErrorCodes.NOT_FOUND, '文件不存在');
  const buf = readFileSync(abs);
  const exam = new ExamTextService();
  try {
    const r = await exam.fromImage(buf);
    return { fileId, text: r.text, confidence: r.confidence, chars: r.text.length };
  } catch (e) {
    return { fileId, text: '', confidence: 0, chars: 0, error: (e as Error).message };
  }
}

```ts
import { Injectable, Logger } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { createChatLlm } from './llm.factory';
import { validateQuestionBank } from './quiz/question-bank-validator';
import type { QuestionBank } from './quiz/question-bank.types';

export interface ExtractResult {
  valid: boolean;
  bank: QuestionBank | null;
  summary: string;
  error?: string;
}

@Injectable()
export class QuestionExtractService {
  private readonly logger = new Logger(QuestionExtractService.name);
  private llm: ReturnType<typeof createChatLlm>;

  constructor(private readonly config: ConfigService) {
    this.llm = createChatLlm(config);
  }

  /** 卷子文本 → 结构化题集（prompt 内嵌 JSON schema + 容错解析 + 校验闭环） */
  async extractQuestions(paperText: string): Promise<ExtractResult> {
    if (!paperText?.trim()) return { valid: false, bank: null, summary: '', error: '卷子文本为空' };
    if (!this.llm) return { valid: false, bank: null, summary: '', error: 'LLM 未配置（demo 模式），无法提取题目' };
    const prompt = this.buildPrompt(paperText);
    try {
      const res = await this.llm.invoke(prompt);
      const content = String(res.content ?? '');
      const bank = this.parseJson(content);
      if (!bank) return { valid: false, bank: null, summary: '', error: 'AI 返回格式不正确（解析失败），请重试' };
      const v = validateQuestionBank(bank);
      if (!v.valid) {
        const codes = [...new Set(v.issues.map((i) => i.code))].join(',');
        return { valid: false, bank, summary: '', error: `题目校验失败（${codes}）：${v.issues[0]?.message ?? ''}` };
      }
      const q = bank.questions;
      const preview = q.slice(0, 3).map((x) => `第${x.no}题(${x.type}) ${x.stem.slice(0, 20)}`).join('；');
      return { valid: true, bank, summary: `已提取 ${q.length} 道题：${preview}${q.length > 3 ? '…' : ''}` };
    } catch (e) {
      this.logger.warn(`题集提取失败: ${(e as Error).message}`);
      return { valid: false, bank: null, summary: '', error: `提取失败：${(e as Error).message}` };
    }
  }

  private buildPrompt(text: string): string {
    return [
      '你是试卷题目解析专家。请从下面的试卷文本中提取所有题目，严格按照 JSON 格式返回，不要包含代码块标记。',
      'JSON 结构：{"paperTitle":"试卷名称","grade":"年级","subject":"科目","questions":[{"no":1,"type":"choice|fill|judge|compute|answer","stem":"题干","options":["A...","B...","C...","D..."],"answer":"答案","analysis":"解析","knowledge":"知识点","points":3}]}',
      '规则：1) type 枚举：choice选择题/fill填空题/judge判断题/compute计算题/answer解答题；2) choice 必须给 options 数组和字母答案（A-F）；3) 题干保留原文；4) 题号从 1 开始连续；5) 无法识别的题目跳过。',
      '试卷文本：',
      text.slice(0, 12000),
    ].join('\n');
  }

  private parseJson(content: string): QuestionBank | null {
    const cleaned = content
      .replace(/```(?:json)?\s*/gi, '')
      .replace(/```/g, '')
      .trim();
    const start = cleaned.indexOf('{');
    const end = cleaned.lastIndexOf('}');
    if (start < 0 || end <= start) return null;
    try {
      return JSON.parse(cleaned.slice(start, end + 1)) as QuestionBank;
    } catch {
      return null;
    }
  }
}

    fs.writeFileSync(path.join(uploadDir, storedName), buffer);
    const record = await this.files.records.save(this.files.records.create({

      storedName,

      ext: 'pptx',
      mime: 'application/vnd.openxmlformats-officedocument.presentationml.presentation',
      size: buffer.length,
      sha256: 'skip',
      uploaderId: user.id,
      category: 'office',
    }));
    return {
      valid: true,
      downloadUrl: `/api/v1/files/${record.id}/download`,

  /** 主题 → 结构模板 + LLM 内容 → 通用 pptx 渲染 → 落盘登记 */
  async generate(user: JwtUser, topic: string, themeKey?: string): Promise<MeetingOutput> {
    const tpl = MEETING_TEMPLATES.find((t) => t.key === themeKey) ?? MEETING_TEMPLATES[0];
    const theme = themeKey && ['default', 'forest', 'ocean', 'sunset', 'ink', 'kids'].includes(themeKey) ? themeKey : tpl.theme;
    if (!this.llm) {
      // demo 降级：用模板结构直接生成空壳 PPT
      return this.renderAndSave(user, this.shellMd(tpl.name, theme));
    }
    const md = await this.askLlm(topic, tpl, theme);
    return this.renderAndSave(user, md);
  }

  private shellMd(title: string, theme: string): string {
    const h = MEETING_TEMPLATES[0].structure.map((s, i) => `# ${s}\n\n- 内容要点 ${i + 1}\n- 内容要点 ${i + 2}`).join('\n\n');
    return `---\ntitle: ${title}\nformat: pptx\ntheme: ${theme}\n---\n\n${h}`;
  }

  private async askLlm(topic: string, tpl: MeetingTemplate, theme: string): Promise<string> {
    const prompt = [
      `你是小学班会课内容专家。请围绕「${topic}」写一份班会 PPT 的 Markdown 内容，严格按以下结构组织，每个环节一个一级标题（#），内容用要点列表（- ），每环节 3-5 个要点，语言适合小学生：`,
      ...tpl.structure.map((s, i) => `${i + 1}. ${s}`),
      '输出格式：Markdown，开头必须是 YAML 头：---\ntitle: <班会标题>\nformat: pptx\n' + `theme: ${theme}\n---`,
      '不要输出代码块标记。',
    ].join('\n');
    const res = await this.llm!.invoke(prompt);
    let md = String(res.content ?? '');
    md = md.replace(/```markdown\s*/gi, '').replace(/```/g, '').trim();
    // 兜底：LLM 忘写 YAML 头时补一个
    if (!md.startsWith('---')) md = `---\ntitle: ${tpl.name}\nformat: pptx\ntheme: ${theme}\n---\n\n${md}`;
    return md;
  }

  private async renderAndSave(user: JwtUser, md: string): Promise<MeetingOutput> {
    const v = validateMarkdown(md);

    return { valid: true, downloadUrl: `/api/v1/files/${record.id}/download`, filename: record.originalName, bytes: record.size };
  }

```ts
@Injectable()
export class OfficeStorageService {
  constructor(
    @InjectRepository(FileRecord) private readonly files: Repository<FileRecord>,
  ) {}
  /** 落盘 uploads/office/ + files 表登记 → 返回 { fileId, downloadUrl, bytes }（与 office.service 现有行为一致） */
  async persist(
    user: { id: number },
    p: { buffer: Buffer; ext: string; mime: string; originalName: string },
  ) {
    const dir = path.join(process.cwd(), 'uploads', 'office');
    fs.mkdirSync(dir, { recursive: true });
    const storedName = `${randomUUID()}.${p.ext}`;
    fs.writeFileSync(path.join(dir, storedName), p.buffer);
    const rec = await this.files.save(
      this.files.create({
        originalName: p.originalName, storedName, size: p.buffer.length,
        mimeType: p.mime, uploaderId: user.id, category: 'document',
        path: path.join(dir, storedName),
      }),
    );
    return { fileId: rec.id, downloadUrl: `/api/v1/files/${rec.id}/download`, bytes: rec.size };
  }
}
